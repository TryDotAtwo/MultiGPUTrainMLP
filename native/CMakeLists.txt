        set(MGT_CUTLASS_USE_SM80 OFF)
        foreach(MGT_CUDA_ARCH IN LISTS CMAKE_CUDA_ARCHITECTURES)
            string(REGEX REPLACE "[^0-9]" "" MGT_CUDA_ARCH_NUM "${MGT_CUDA_ARCH}")
            if(MGT_CUDA_ARCH_NUM GREATER_EQUAL 80)
                set(MGT_CUTLASS_USE_SM80 ON)
            endif()
        endforeach()
        target_compile_definitions(mgt_cuda PUBLIC MGT_HAS_CUTLASS=1)
        set(MGT_CUTLASS_HALF_GEMM_KIND_MASK 0)
        string(REPLACE "," ";" MGT_CUTLASS_HALF_GEMM_KIND_LIST "${MGT_CUTLASS_HALF_GEMM_KINDS}")
        foreach(MGT_CUTLASS_KIND IN LISTS MGT_CUTLASS_HALF_GEMM_KIND_LIST)
            string(STRIP "${MGT_CUTLASS_KIND}" MGT_CUTLASS_KIND)
            string(TOLOWER "${MGT_CUTLASS_KIND}" MGT_CUTLASS_KIND)
            if(MGT_CUTLASS_KIND STREQUAL "" OR MGT_CUTLASS_KIND STREQUAL "none")
                continue()
            elseif(MGT_CUTLASS_KIND STREQUAL "all")
                math(EXPR MGT_CUTLASS_HALF_GEMM_KIND_MASK "${MGT_CUTLASS_HALF_GEMM_KIND_MASK} | 15")
            elseif(MGT_CUTLASS_KIND STREQUAL "forward")
                math(EXPR MGT_CUTLASS_HALF_GEMM_KIND_MASK "${MGT_CUTLASS_HALF_GEMM_KIND_MASK} | 1")
            elseif(MGT_CUTLASS_KIND STREQUAL "grad_weights")
                math(EXPR MGT_CUTLASS_HALF_GEMM_KIND_MASK "${MGT_CUTLASS_HALF_GEMM_KIND_MASK} | 2")
            elseif(MGT_CUTLASS_KIND STREQUAL "backprop_input")
                math(EXPR MGT_CUTLASS_HALF_GEMM_KIND_MASK "${MGT_CUTLASS_HALF_GEMM_KIND_MASK} | 4")
            elseif(MGT_CUTLASS_KIND STREQUAL "input_embedding_grad")
                math(EXPR MGT_CUTLASS_HALF_GEMM_KIND_MASK "${MGT_CUTLASS_HALF_GEMM_KIND_MASK} | 8")
            else()
                message(FATAL_ERROR "Unknown MGT_CUTLASS_HALF_GEMM_KINDS entry: ${MGT_CUTLASS_KIND}")
            endif()
        endforeach()
        target_compile_definitions(mgt_cuda PUBLIC MGT_CUTLASS_HALF_GEMM_KIND_MASK=${MGT_CUTLASS_HALF_GEMM_KIND_MASK})
        if(MGT_USE_CUTLASS_HALF_GEMM OR MGT_AUTO_CUTLASS_HALF_GEMM)
            target_compile_definitions(mgt_cuda PUBLIC MGT_HAS_CUTLASS_HALF_GEMM=1)
        endif()
        if(MGT_USE_CUTLASS_HALF_GEMM)
            target_compile_definitions(mgt_cuda PUBLIC MGT_USE_CUTLASS_HALF_GEMM=1)
        elseif(MGT_AUTO_CUTLASS_HALF_GEMM)
            target_compile_definitions(mgt_cuda PUBLIC MGT_AUTO_CUTLASS_HALF_GEMM=1)
        endif()
        if(MGT_CUTLASS_USE_SM80)
            target_compile_definitions(mgt_cuda PUBLIC MGT_CUTLASS_ARCH_SM80=1)
            set(MGT_CUTLASS_ARCH_MESSAGE "Sm80")
        else()
            target_compile_definitions(mgt_cuda PUBLIC MGT_CUTLASS_ARCH_SM75=1)
            set(MGT_CUTLASS_ARCH_MESSAGE "Sm75")
        endif()
        if(MGT_USE_CUTLASS_HALF_GEMM)
            message(STATUS "CUTLASS found: ${CUTLASS_INCLUDE_DIR}; half GEMM backend=CUTLASS ${MGT_CUTLASS_ARCH_MESSAGE} tensorop kernels; kinds=all")
        elseif(MGT_AUTO_CUTLASS_HALF_GEMM)
            message(STATUS "CUTLASS found: ${CUTLASS_INCLUDE_DIR}; half GEMM backend=AUTO ${MGT_CUTLASS_ARCH_MESSAGE} tensorop candidates + cuBLASLt default; kinds=${MGT_CUTLASS_HALF_GEMM_KINDS}; mask=${MGT_CUTLASS_HALF_GEMM_KIND_MASK}")
        else()
            message(STATUS "CUTLASS found: ${CUTLASS_INCLUDE_DIR}; half GEMM backend=cuBLASLt; CUTLASS backend available with -DMGT_USE_CUTLASS_HALF_GEMM=ON")
        endif()
    elseif(MGT_USE_CUTLASS_HALF_GEMM)
        message(FATAL_ERROR "MGT_USE_CUTLASS_HALF_GEMM=ON but CUTLASS was not found. Set MGT_CUTLASS_ROOT or CUTLASS_ROOT.")
